🔧 SPECIFICATIONS TECHNIQUES - Bridge

Problématique

Express.js (JavaScript) ≠ Algorithmes (Python)
→ Besoin de communication inter-processus

Solution: pythonBridge.js

const { spawn } = require('child_process'); function generateMaze(largeur, hauteur) { return new Promise((resolve, reject) => { const python = spawn('python', [ 'main.py', largeur, hauteur ]); let dataString = ''; python.stdout.on('data', (data) => { dataString += data.toString(); }); python.on('close', (code) => { if (code === 0) { resolve(JSON.parse(dataString)); } else { reject('Python error'); } }); }); }
